cmake_minimum_required(VERSION 3.16)
project(fabtwin_edge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WITH_KAFKA "Build librdkafka transport" OFF)
find_package(Threads REQUIRED)
add_executable(fabtwin-edge main.cpp simulator/Machine.cpp agent/EdgeAgent.cpp agent/NetworkClient.cpp)
target_include_directories(fabtwin-edge PRIVATE .)
target_link_libraries(fabtwin-edge PRIVATE Threads::Threads)
if(WITH_KAFKA)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(RDKAFKA REQUIRED IMPORTED_TARGET rdkafka)
  target_compile_definitions(fabtwin-edge PRIVATE WITH_KAFKA)
  target_link_libraries(fabtwin-edge PRIVATE PkgConfig::RDKAFKA)
endif()
if(MSVC)
  target_compile_options(fabtwin-edge PRIVATE /W4)
else()
  target_compile_options(fabtwin-edge PRIVATE -Wall -Wextra -Wpedantic)
endif()
